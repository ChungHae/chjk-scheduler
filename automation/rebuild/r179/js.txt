  function _mmMatchVendor(tokens){
    var names=allClients().map(function(c){ return String(c[0]||'').trim(); }).filter(Boolean);
    var lower={}; names.forEach(function(n){ lower[n.toLowerCase()]=n; });
    for(var n=Math.min(3,tokens.length); n>=1; n--){
      var cand=tokens.slice(0,n).join(' ').toLowerCase();
      if(lower[cand]) return { vendor:lower[cand], used:n };
      var cand2=tokens.slice(0,n).join('').toLowerCase();
      if(lower[cand2]) return { vendor:lower[cand2], used:n };
    }
    var t0=String(tokens[0]||'').toLowerCase(); if(t0.length<2) return null;
    var hits=names.filter(function(nm){ var l=nm.toLowerCase(); return l.indexOf(t0)===0 || (t0.length>=3 && t0.indexOf(l)===0); });
    if(hits.length===1) return { vendor:hits[0], used:1 };
    if(hits.length>1){ hits.sort(function(a,b){ return a.length-b.length; }); return { vendor:hits[0], used:1, ambiguous:true }; }
    return null;
  }

  function _mmParseWhen(text){
    var out={ date:'', time:'' };
    var m=text.match(/(\d{1,2})\s*[\/.월]\s*(\d{1,2})\s*일?/);
    if(m){ var mo=+m[1], d=+m[2]; if(mo>=1&&mo<=12&&d>=1&&d<=31){ var now=new Date(); var y=now.getFullYear(); if(mo<now.getMonth()+1-6) y++; out.date=y+'-'+String(mo).padStart(2,'0')+'-'+String(d).padStart(2,'0'); } }
    var t=text.match(/(\d{1,2})\s*:\s*(\d{2})/);
    if(t){ out.time=String(Math.min(23,+t[1])).padStart(2,'0')+':'+t[2]; }
    else { var t2=text.match(/(오전|오후)?\s*(\d{1,2})\s*시\s*(반|(\d{1,2})\s*분)?/); if(t2){ var h=+t2[2]; if(t2[1]==='오후'&&h<12) h+=12; if(!t2[1]&&h<8) h+=12; var mi=t2[3]==='반'?30:(t2[4]?+t2[4]:0); out.time=String(Math.min(23,h)).padStart(2,'0')+':'+String(mi).padStart(2,'0'); } }
    return out;
  }
  var _MM_PHONE=/(0\d{1,2})[-\s.]?(\d{3,4})[-\s.]?(\d{4})/;
  // ── 한 줄 해석 ──
  function _mmParse(raw){
    var text=String(raw||'').trim();
    var r={ text:text, kind:'memo', vendor:'', to:'', name:'', phone:'', when:{date:'',time:''}, ambiguous:false };
    if(!text) return r;
    var tokens=text.split(/\s+/);
    var k0=tokens[0].replace(/[:：]$/,'');
    var kind=_mmKW[k0];
    if(kind){
      r.kind=kind;
      var rest=tokens.slice(1);
      if(kind==='request'){
        var to=(rest[0]||'').replace(/(님|에게|한테)$/,'');
        if(_mmMemberByName(to)){ r.to=to; }
        else { r.kind='memo'; }   // 등록된 이름이 아니면 그냥 메모
      } else if(kind!=='notice'){
        var mv=rest.length?_mmMatchVendor(rest):null;
        if(mv){ r.vendor=mv.vendor; r.ambiguous=!!mv.ambiguous; }
        r.when=_mmParseWhen(rest.join(' '));
      }
      return r;
    }
    var ph=text.match(_MM_PHONE);
    if(ph){
      r.kind='contact'; r.phone=ph[1]+'-'+ph[2]+'-'+ph[3];
      var nm=text.slice(0, ph.index).replace(/[\s,·:：\-]+$/,'').trim();
      r.name = nm.split(/\s+/).pop() || '';
      return r;
    }
    return r;
  }
